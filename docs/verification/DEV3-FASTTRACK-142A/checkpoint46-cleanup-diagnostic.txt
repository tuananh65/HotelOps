TR_hk_defects_v1 CREATE TRIGGER dbo.TR_hk_defects_v1 ON dbo.inspection_defects AFTER INSERT,UPDATE,DELETE AS
BEGIN
 SET NOCOUNT ON;
 IF EXISTS(SELECT 1 FROM inserted x JOIN dbo.room_inspections i ON i.inspection_id=x.inspection_id
  JOIN dbo.fe05_reinspection_bindings_v1 b ON b.inspection_id=i.inspection_id WHERE i.workflow_status='ASSIGNED'
  AND NOT EXISTS(SELECT 1 FROM dbo.fe05_reinspection_defect_results_v1 v JOIN dbo.inspection_defects original ON original.defect_id=v.defect_id
    WHERE v.inspection_id=i.inspection_id AND v.rework_id=b.rework_id AND v.fixed=0 AND v.recorded_by=i.inspector_user_id
      AND original.checklist_item_id=x.checklist_item_id AND original.description=x.description))
  THROW 51999,'Remaining defect must come from a Not Fixed decision',1;
 IF EXISTS(SELECT 1 FROM (SELECT inspection_id FROM inserted UNION SELECT inspection_id FROM deleted) a JOIN dbo.room_inspections r WITH(UPDLOCK,HOLDLOCK) ON r.inspection_id=a.inspection_id
  WHERE r.workflow_status IS NOT NULL AND (r.workflow_status NOT IN('INSPECTING','PENDING_CONCLUSION') AND NOT(r.workflow_status='ASSIGNED' AND EXISTS(SELECT 1 FROM dbo.fe05_reinspection_bindings_v1 b
 WHERE b.inspection_id=r.inspection_id AND b.reinspector_user_id=r.inspector_user_id))))
  THROW 51731,'Concluded inspection defects cannot change.',1;
 IF EXISTS(SELECT 1 FROM inserted i JOIN dbo.room_inspections r ON r.inspection_id=i.inspection_id
  WHERE r.workflow_status IS NOT NULL AND (i.checklist_item_id IS NULL OR LEN(LTRIM(RTRIM(i.description)))=0
   OR NOT EXISTS(SELECT 1 FROM dbo.inspection_results x JOIN dbo.inspection_checklist_items c ON c.item_id=x.checklist_item_id
    WHERE x.inspection_id=r.inspection_id AND x.checklist_item_id=i.checklist_item_id AND x.outcome='FAIL' AND c.version_id=r.version_id)))
  THROW 51731,'Defect requires a failed criterion from this inspection.',1;
 UPDATE r SET summary_note=summary_note FROM dbo.room_inspections r JOIN
  (SELECT inspection_id FROM inserted UNION SELECT inspection_id FROM deleted) a ON a.inspection_id=r.inspection_id WHERE r.workflow_status IS NOT NULL;
END

TR_hk_results_v1 CREATE TRIGGER dbo.TR_hk_results_v1 ON dbo.inspection_results AFTER INSERT,UPDATE,DELETE AS
BEGIN
 SET NOCOUNT ON;
 IF EXISTS(SELECT 1 FROM inserted x JOIN dbo.room_inspections i ON i.inspection_id=x.inspection_id
  JOIN dbo.fe05_reinspection_bindings_v1 b ON b.inspection_id=i.inspection_id WHERE i.workflow_status='ASSIGNED' AND
   (NOT EXISTS(SELECT 1 FROM dbo.fe05_reinspection_defect_results_v1 v JOIN dbo.inspection_defects d ON d.defect_id=v.defect_id
      WHERE v.inspection_id=i.inspection_id AND v.rework_id=b.rework_id AND d.checklist_item_id=x.checklist_item_id)
    OR EXISTS(SELECT 1 FROM dbo.fe05_reinspection_defect_results_v1 v JOIN dbo.inspection_defects d ON d.defect_id=v.defect_id
      WHERE v.inspection_id=i.inspection_id AND d.checklist_item_id=x.checklist_item_id AND(v.fixed IS NULL OR v.recorded_by<>i.inspector_user_id))
    OR x.outcome<>CASE WHEN EXISTS(SELECT 1 FROM dbo.fe05_reinspection_defect_results_v1 v JOIN dbo.inspection_defects d ON d.defect_id=v.defect_id
      WHERE v.inspection_id=i.inspection_id AND d.checklist_item_id=x.checklist_item_id AND v.fixed=0) THEN 'FAIL' ELSE 'PASS' END))
  THROW 51999,'Reinspection criterion projection must match exact defect decisions',1;
 IF EXISTS(SELECT 1 FROM (SELECT inspection_id FROM inserted UNION SELECT inspection_id FROM deleted) a
  JOIN dbo.room_inspections r WITH(UPDLOCK,HOLDLOCK) ON r.inspection_id=a.inspection_id WHERE r.workflow_status IS NOT NULL AND (r.workflow_status<>'INSPECTING' AND NOT(r.workflow_status='ASSIGNED' AND EXISTS(SELECT 1 FROM dbo.fe05_reinspection_bindings_v1 b
 WHERE b.inspection_id=r.inspection_id AND b.reinspector_user_id=r.inspector_user_id))))
  THROW 51731,'Results writable only while Inspecting.',1;
 IF EXISTS(SELECT 1 FROM inserted i JOIN dbo.room_inspections r ON r.inspection_id=i.inspection_id JOIN dbo.inspection_checklist_items c ON c.item_id=i.checklist_item_id
  WHERE r.workflow_status IS NOT NULL AND (c.version_id<>r.version_id OR c.is_active=0 OR
   EXISTS(SELECT 1 FROM dbo.inspection_results x WITH(UPDLOCK,HOLDLOCK) WHERE x.inspection_id=i.inspection_id AND x.checklist_item_id=i.checklist_item_id AND x.result_id<>i.result_id)))
  THROW 51731,'Result must be unique and belong to assigned active criterion.',1;
 UPDATE r SET summary_note=summary_note FROM dbo.room_inspections r JOIN
  (SELECT inspection_id FROM inserted UNION SELECT inspection_id FROM deleted) a ON a.inspection_id=r.inspection_id WHERE r.workflow_status IS NOT NULL;
END
