foreach($path in $additions){if(($intended+$newFiles) -notcontains $path){$issues.Add('Unexpected addition/change: '+$path)}}
foreach($path in $newFiles){if(!(Test-Path -LiteralPath $path)){$issues.Add('Missing expected new file: '+$path)}}
$indexUnchanged=@(Compare-Object @($before.IndexEntries) @(git ls-files --stage)).Count -eq 0

 $path=$base+$doc;$content=[IO.File]::ReadAllText((Join-Path $repoRoot $path))
 foreach($match in [regex]::Matches($content,'\[[^\]\r\n]+\]\(([^\)\r\n]+)\)')){
  $target=$match.Groups[1].Value.Trim('<','>');if($target -match '^https?://' -or $target.StartsWith('#')){continue}
  $target=($target -split '#',2)[0];$resolved=[IO.Path]::GetFullPath((Join-Path (Split-Path (Join-Path $repoRoot $path)) $target))
  if(!(Test-Path -LiteralPath $resolved)){$issues.Add('Broken link: '+$doc+' => '+$target)};$links++
 }
}
foreach($path in ($intended+$newFiles | Where-Object {$_ -notlike '*.json'})){
 $content=[IO.File]::ReadAllText((Join-Path $repoRoot $path))
 if($content -match '(?m)[ \t]+\r?$'){$issues.Add('Trailing whitespace: '+$path)}
 if($content -match '(?m)^(<<<<<<< |=======\s*$|>>>>>>> )'){$issues.Add('Conflict marker: '+$path)}
}
$accepted=Get-Content -LiteralPath ($base+'task58-writers-accepted.local.json') -Raw -Encoding UTF8 | ConvertFrom-Json

$result=[ordered]@{

$result | ConvertTo-Json -Depth 7
