async Task<string> Snapshot()
{
    // Compare all mapped rows only in memory. No credentials/personal snapshots in receipts/logs.
    await using var db = Open(); await db.Database.OpenConnectionAsync(ct);
    static string Q(string name) => "[" + name.Replace("]", "]]") + "]";
    var rows = new List<string>();
    foreach (var entity in db.Model.GetEntityTypes().Where(e => e.GetTableName() != null).OrderBy(e => e.GetTableName()))
    {
        var table = entity.GetTableName()!; var store = StoreObjectIdentifier.Table(table, entity.GetSchema());
        var keys = entity.FindPrimaryKey()!.Properties.Select(p => Q(p.GetColumnName(store)!));
        await using var cmd = db.Database.GetDbConnection().CreateCommand(); cmd.CommandTimeout = 20;
        cmd.CommandText = "SELECT COALESCE((SELECT * FROM " + Q(entity.GetSchema() ?? "dbo") + "." + Q(table)
            + " ORDER BY " + string.Join(",", keys) + " FOR JSON PATH,INCLUDE_NULL_VALUES),N'[]')";
        rows.Add(table + ":" + (string)(await cmd.ExecuteScalarAsync(ct))!);
    }
    return JsonSerializer.Serialize(rows);
}

static bool HasTimeout(Exception error)
{
    // EF's SQL execution strategy can wrap a TimeoutException in InvalidOperationException.
    // Accept the injected provider fault only; unrelated programming errors must still fail.
    for (Exception? inner = error; inner != null; inner = inner.InnerException)
        if (inner is TimeoutException { Message: "Task60 deliberate query-command fault" }) return true;
    return false;
}

sealed class Clock(DateTime now) : TimeProvider
{
    public DateTime Now { get; set; } = now;
    public override DateTimeOffset GetUtcNow() => new(Now, TimeSpan.Zero);
}
sealed class Probe : DbCommandInterceptor
{
    public int Commands { get; private set; }
    public int CountQueries { get; private set; }
    public int PageQueries { get; private set; }
    public int RoomQueries => CountQueries + PageQueries;
    public bool PausePage { get; init; }
    public string? Failure { get; init; }
    public TaskCompletionSource Entered { get; } = new(TaskCreationOptions.RunContinuationsAsynchronously);
    public TaskCompletionSource Release { get; } = new(TaskCreationOptions.RunContinuationsAsynchronously);
    public override async ValueTask<InterceptionResult<DbDataReader>> ReaderExecutingAsync(DbCommand command,
        CommandEventData eventData, InterceptionResult<DbDataReader> result, CancellationToken cancellationToken = default)
    {
        Commands++;
        if (command.CommandText.Contains("UC023:Count")) CountQueries++;
        if (command.CommandText.Contains("UC023:Page"))
        {
            PageQueries++;
            if (Failure == "page") throw new TimeoutException("Task60 deliberate query-command fault");
            if (PausePage) { Entered.TrySetResult(); await Release.Task.WaitAsync(cancellationToken); }
        }
        if (Failure == "authority" && command.CommandText.Contains("[role_permissions]"))
            throw new TimeoutException("Task60 deliberate authority-command fault");
        return result;
    }
}
