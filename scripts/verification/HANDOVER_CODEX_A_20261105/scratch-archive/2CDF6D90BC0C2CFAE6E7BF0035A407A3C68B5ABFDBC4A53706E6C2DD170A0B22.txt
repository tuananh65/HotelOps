async Task<string> Fingerprint(params string[] tables)
{
    var hashes = new List<string>();
    foreach (var table in tables)
    {
        await using var cmd = db.Database.GetDbConnection().CreateCommand();
        // Real credential logins update LastLoginAt; deliberate actor-state fault fixtures
        // advance rowversion. Every other column is compared here; POST-only full-row proof below.
        var columns = table == "users" ? string.Join(',', db.Model.FindEntityType(typeof(User))!.GetProperties()
            .Select(x => x.GetColumnName(StoreObjectIdentifier.Table("users", null))!)
            .Where(x => x is not "row_version" and not "last_login_at").Select(x => "[" + x.Replace("]", "]]") + "]")) : "*";
        var order = table == "users" ? "user_id" : "1";
        cmd.CommandText = $"SELECT {columns} FROM dbo.[{table}] ORDER BY {order} FOR JSON PATH,INCLUDE_NULL_VALUES";
        await using var reader = await cmd.ExecuteReaderAsync(ct); var text = new StringBuilder();
        while (await reader.ReadAsync(ct)) text.Append(reader.GetString(0));
        hashes.Add(Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(text.ToString()))));
    }
    return string.Join(':', hashes);
}

internal sealed class CookieJar
{
    public Dictionary<string, string> Cookies { get; } = new();
    public string? Token { get; set; }
    public void Read(HttpResponseMessage response)
    {
        if (response.Headers.TryGetValues("Set-Cookie", out var values)) foreach (var value in values)
        { var pair = value.Split(';')[0].Split('=', 2); if (pair.Length == 2) Cookies[pair[0]] = pair[1]; }
    }
}
