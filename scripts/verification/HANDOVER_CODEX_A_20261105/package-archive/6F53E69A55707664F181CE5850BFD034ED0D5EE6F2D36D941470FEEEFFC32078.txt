   72:                     && a.Shift.Status == "PLANNED" && a.Shift.ShiftDate == today

   83:     private async Task<WorkdayResult> Execute(long branchId, string permission,
   84:         Func<CurrentUserInfo, Task<WorkdayResult>> action, CancellationToken ct)
   85:     {
   86:         if (!current.TryGetCurrentUser(out var actor) || actor.UserId <= 0 || actor.SessionId <= 0)
   87:             return new(401, Code: "SESSION_INVALID");
   88:         if (actor.IsRestricted) return new(403, Code: "PASSWORD_CHANGE_REQUIRED");
   89:         if (branchId <= 0) return new(403, Code: "TIME_SCOPE_DENIED");
   90:         var context = http.HttpContext;
   91:         if (context is null || !HotelOpsCookieTicket.TryRead(context.User, out var ticket, out var sessionId)
   92:             || ticket is null || sessionId != actor.SessionId || ticket.UserId != actor.UserId)
   93:             return new(401, Code: "SESSION_INVALID");
   94:         try
   95:         {
   96:             // Serializable retains session/grant/source reads until commit. User/time locks first;
   97:             // stale middleware identity cannot survive a revocation committed during a lock wait.
   98:             await using var tx = await db.Database.BeginTransactionAsync(IsolationLevel.Serializable, ct);
   99:             await participant.LockUsersAsync([actor.UserId], ct);
  100:             var validated = await sessions.ValidateSessionAsync(ticket, ct);
  101:             if (!validated.IsValid || validated.Session?.SessionId != actor.SessionId
  102:                 || validated.Session.UserId != actor.UserId || validated.Session.IsRestricted)
  103:                 return new(401, Code: "SESSION_INVALID");
  104:             if (!await db.Users.AsNoTracking().AnyAsync(u => u.UserId == actor.UserId
  105:                     && u.AccountStatus == "ACTIVE" && u.EmploymentStatus == "ACTIVE" && u.DeletedAt == null, ct))
  106:                 return new(403, Code: "TIME_ACTOR_INVALID");
  107:             var denied = await Authorize(permission, new("workday_self_branch", branchId), ct);
  108:             if (denied is not null) return denied;
  109:             var result = await action(actor);
  110:             if (result.Status is >= 200 and < 300) await tx.CommitAsync(ct);
  111:             return result;
  112:         }
  113:         catch (WorkTimeContractException e)
  114:         {
  115:             return e.Code switch
  116:             {
  117:                 "TIME_STALE" => new(412, Code: e.Code),
  118:                 "TIME_SCOPE_DENIED" or "TIME_ACTOR_INVALID" => new(403, Code: e.Code),
  119:                 "TIME_SESSION_INVALID" => new(401, Code: "SESSION_INVALID"),
  120:                 "TIME_RESTRICTED" => new(403, Code: "PASSWORD_CHANGE_REQUIRED"),
  121:                 "TIME_SOURCE_UNRESOLVED" or "TIME_BUSY" => new(503, Code: e.Code),
  122:                 "TIME_INTERVAL_ALREADY_OPEN" => new(409, Code: "MSG31_PAUSE_ALL_REQUIRED"),
  123:                 _ => new(409, Code: e.Code)
  124:             };
  125:         }
  126:         catch (SessionStoreUnavailableException) { return new(503, Code: "SESSION_STORE_UNAVAILABLE"); }
  127:         catch (SqlException) when (!ct.IsCancellationRequested) { return new(503, Code: "WORKDAY_STORE_UNAVAILABLE"); }
  128:         catch (DbUpdateException) when (!ct.IsCancellationRequested) { return new(503, Code: "WORKDAY_STORE_UNAVAILABLE"); }
  129:     }
