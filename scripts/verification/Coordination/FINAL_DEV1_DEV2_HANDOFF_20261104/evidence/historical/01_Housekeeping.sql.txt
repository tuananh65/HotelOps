43: CREATE TRIGGER dbo.TR_hk_inspections_v1 ON dbo.room_inspections AFTER INSERT,UPDATE AS
44: BEGIN
45:  SET NOCOUNT ON;
46:  IF EXISTS(SELECT 1 FROM inserted i JOIN deleted d ON d.inspection_id=i.inspection_id WHERE d.workflow_status IS NOT NULL AND
47:   (i.workflow_status IS NULL OR i.room_id<>d.room_id OR ISNULL(i.cleaning_task_id,0)<>ISNULL(d.cleaning_task_id,0) OR i.version_id<>d.version_id))
48:   THROW 51731,'Owned inspection source/version is immutable.',1;
49:  IF EXISTS(SELECT 1 FROM inserted i LEFT JOIN dbo.cleaning_tasks t WITH(HOLDLOCK) ON t.cleaning_task_id=i.cleaning_task_id
50:   WHERE i.workflow_status IS NOT NULL AND (t.cleaning_task_id IS NULL OR t.room_id<>i.room_id))
51:   THROW 51731,'Inspection must reference its own cleaning cycle.',1;
52:  IF EXISTS(SELECT 1 FROM inserted i WHERE i.workflow_status IS NOT NULL AND i.status IN('PENDING','IN_PROGRESS')
53:   AND EXISTS(SELECT 1 FROM dbo.room_inspections other WITH(UPDLOCK,HOLDLOCK) WHERE other.cleaning_task_id=i.cleaning_task_id AND other.inspection_id<>i.inspection_id AND other.status IN('PENDING','IN_PROGRESS')))
54:   THROW 51731,'Only one open inspection per cleaning cycle.',1;
55: END

58: BEGIN
59:  SET NOCOUNT ON;
60:  IF EXISTS(SELECT 1 FROM (SELECT inspection_id FROM inserted UNION SELECT inspection_id FROM deleted) a
