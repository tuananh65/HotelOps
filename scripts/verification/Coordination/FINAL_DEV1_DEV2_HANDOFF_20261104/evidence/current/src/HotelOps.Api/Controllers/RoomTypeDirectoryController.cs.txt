30:         {

33:             {
34:                 if (result.ETag != null) Response.Headers.ETag = result.ETag;
35:                 return Ok(result.Value);

40:                 RoomTypeDirectoryStatus.NotFound => 404, RoomTypeDirectoryStatus.Inactive => 409, _ => 503
