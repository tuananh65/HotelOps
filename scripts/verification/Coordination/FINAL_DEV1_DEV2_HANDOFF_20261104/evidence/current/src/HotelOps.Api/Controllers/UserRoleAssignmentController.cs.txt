11: [ApiController]
12: [Route("api/users")]
13: [Authorize(AuthenticationSchemes = HotelOpsCookieTicket.Scheme)]
14: [ResponseCache(NoStore = true, Location = ResponseCacheLocation.None)]
15: [UserAccountActivationErrors]
16: public sealed class UserRoleAssignmentController(UserRoleAssignmentService assignments) : ControllerBase
17: {
18:     [HttpPost("{userId}/roles")]
19:     [RequestSizeLimit(16 * 1024)]
20:     public async Task<IActionResult> Assign([FromRoute, Range(1, long.MaxValue)] long userId, [FromBody] JsonElement body, CancellationToken ct)
21:     {
22:         if (!Request.IsHttps) return Problem(statusCode: 400, title: "HTTPS is required");
23:         if (!HotelOpsCookieTicket.TryRead(User, out var reference, out var sessionId)) return Failure(401, "SESSION_INVALID");
24:         if (!Request.Headers.TryGetValue("If-Match", out var header)) return Failure(428, "IF_MATCH_REQUIRED");
25:         var tag = header.ToString();
28:         if (!AssignUserRoleRequest.TryParse(body, out var request)) return Failure(400, "VALIDATION_FAILED");
29:         var result = await assignments.AssignAsync(reference!, sessionId, userId, request!, Convert.FromHexString(tag[1..^1]), ct);
30:         switch (result.Status)
31:         {
32:             case AssignRoleStatus.Success:
33:                 Response.Headers.ETag = "\"" + Convert.ToHexString(result.Version!) + "\"";
34:                 return StatusCode(201, result.Assignment);
35:             case AssignRoleStatus.SessionInvalid: return Failure(401, "SESSION_INVALID");
36:             case AssignRoleStatus.Forbidden: return Failure(403, "ROLE_ASSIGNMENT_NOT_ALLOWED");
37:             case AssignRoleStatus.NotFound: return Failure(404, "USER_NOT_FOUND");
38:             case AssignRoleStatus.Inactive: return Failure(409, "TARGET_NOT_ACTIVE");
39:             case AssignRoleStatus.Stale: return Failure(412, "STALE_USER_VERSION");
40:             case AssignRoleStatus.ScopeConflict: return Failure(409, "PRIMARY_SCOPE_CONFLICT");
41:             case AssignRoleStatus.RoleConflict: return Failure(409, "EXISTING_ROLE_PATH_CONFLICT");
42:             case AssignRoleStatus.LastAdministrator: return Failure(409, "LAST_DURABLE_ADMIN");
43:             case AssignRoleStatus.SerializationUnavailable: return Failure(503, "GLOBAL_ADMIN_SERIALIZATION_UNAVAILABLE");
44:             default: throw new InvalidOperationException("Unknown role assignment outcome.");
45:         }
46:     }
47:     private ObjectResult Failure(int status, string code) => Problem(statusCode: status, title: code,
48:         extensions: new Dictionary<string, object?> { ["code"] = code });
49: }
