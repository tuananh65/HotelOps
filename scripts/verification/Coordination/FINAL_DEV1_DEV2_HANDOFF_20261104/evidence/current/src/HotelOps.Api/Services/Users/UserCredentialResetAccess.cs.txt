9: /// employee identity verification. The reset workflow combines all these gates.
10: /// </summary>
11: public sealed class UserCredentialResetAccess(HotelOpsDbContext db, ICurrentUserAccessor current,
12:     IPermissionService permissions, IPermissionDataSource data, TimeProvider clock)
13: {
14:     /// <summary>Retain relevant authorization rows and key ranges until the reset transaction commits.</summary>
15:     internal async Task LockAuthorityAsync(long actorId, long targetId, CancellationToken ct)
16:     {
17:         if (db.Database.CurrentTransaction is null || !db.Database.IsSqlServer())
18:             throw new InvalidOperationException("Reset authority locks require the caller SQL transaction.");
19:         // Lock all paths, including inactive/absent ranges, before evaluating them. Ordinary
20:         // SQL UPDATE/DELETE/INSERT writers conflict with HOLDLOCK without a new writer protocol.
21:         _ = await db.UserRoles.FromSqlInterpolated($"SELECT * FROM dbo.user_roles WITH (HOLDLOCK) WHERE user_id={actorId}")
22:             .AsNoTracking().ToListAsync(ct);
23:         _ = await db.Roles.FromSqlInterpolated($"SELECT * FROM dbo.roles r WITH (HOLDLOCK) WHERE EXISTS (SELECT 1 FROM dbo.user_roles ur WHERE ur.user_id={actorId} AND ur.role_id=r.role_id)")
24:             .AsNoTracking().ToListAsync(ct);
25:         _ = await db.RolePermissions.FromSqlInterpolated($"SELECT * FROM dbo.role_permissions rp WITH (HOLDLOCK) WHERE EXISTS (SELECT 1 FROM dbo.user_roles ur WHERE ur.user_id={actorId} AND ur.role_id=rp.role_id)")
26:             .AsNoTracking().ToListAsync(ct);
27:         _ = await db.Permissions.FromSqlInterpolated($"SELECT * FROM dbo.permissions WITH (HOLDLOCK) WHERE permission_code={"USER_MANAGE"}")
28:             .AsNoTracking().ToListAsync(ct);
29:         foreach (var id in new[] { actorId, targetId }.Order())
30:             _ = await db.UserBranchAccesses.FromSqlInterpolated($"SELECT * FROM dbo.user_branch_access WITH (HOLDLOCK) WHERE user_id={id}")
31:                 .AsNoTracking().ToListAsync(ct);
32:         _ = await db.Branches.FromSqlInterpolated($"SELECT * FROM dbo.branches b WITH (HOLDLOCK) WHERE EXISTS (SELECT 1 FROM dbo.user_branch_access a WHERE a.user_id IN ({actorId},{targetId}) AND a.branch_id=b.branch_id) OR EXISTS (SELECT 1 FROM dbo.user_roles ur WHERE ur.user_id={actorId} AND ur.branch_id=b.branch_id)")
33:             .AsNoTracking().ToListAsync(ct);
34:     }
35: 
36:     public async Task<bool> CanResetAsync(long targetId, CancellationToken ct)
37:     {
38:         if (targetId <= 0 || !current.TryGetCurrentUser(out var actor) || actor.IsRestricted
39:             || actor.UserId == targetId) return false;
41:         var now = clock.GetUtcNow().UtcDateTime;
42:         var branches = await db.UserBranchAccesses.AsNoTracking()
43:             .Where(a => a.UserId == actor.UserId && a.RevokedAt == null && a.GrantedAt <= now && a.Branch.IsActive
44:                 && db.UserBranchAccesses.Any(t => t.UserId == targetId && t.RevokedAt == null && t.GrantedAt <= now
45:                     && t.BranchId == a.BranchId && t.User.DeletedAt == null))
46:             .Select(a => a.BranchId).Distinct().OrderBy(id => id).ToArrayAsync(ct);
47:         var evaluator = new PermissionService(current, new BranchOnlyData(data),
48:             new UnresolvedPermissionResourceResolver(), clock);
49:         foreach (var branch in branches)
50:             if ((await evaluator.EvaluateBranchAsync("USER_MANAGE", branch, ct)).IsGranted) return true;
51:         return false;
52:     }
53: 
54:     private sealed class BranchOnlyData(IPermissionDataSource source) : IPermissionDataSource
55:     {
56:         public Task<bool> IsActiveUserAsync(long id, CancellationToken ct) => source.IsActiveUserAsync(id, ct);
57:         public Task<bool> IsActiveBranchAsync(long id, CancellationToken ct) => source.IsActiveBranchAsync(id, ct);
58:         public async Task<IReadOnlyList<PermissionGrantPath>> ReadPathsAsync(long id, string code, DateTime now, CancellationToken ct)
59:             => (await source.ReadPathsAsync(id, code, now, ct)).Where(p => p.DataScope == "BRANCH").ToArray();
60:     }
61: }
