14: [Authorize(AuthenticationSchemes = HotelOpsCookieTicket.Scheme)]
15: [ResponseCache(NoStore = true, Location = ResponseCacheLocation.None)]
16: public sealed class AdminCredentialResetController(AdminCredentialResetWorkflow resets) : ControllerBase
17: {
18:     [HttpPost("{userId}/reset-credential")]
19:     public async Task<IActionResult> Reset([FromRoute, Range(1, long.MaxValue)] long userId,
20:         [FromBody] JsonElement body, CancellationToken ct)
21:     {
22:         if (!Request.IsHttps) return Problem(statusCode: 400, title: "HTTPS is required");
23:         if (!HotelOpsCookieTicket.TryRead(User, out var reference, out var sessionId)) return Failure(401, "SESSION_INVALID");
24:         if (!Request.Headers.TryGetValue("If-Match", out var header)) return Failure(428, "IF_MATCH_REQUIRED");
25:         var tag = header.ToString();
32:             return Failure(400, "IDENTITY_CONFIRMATION_REQUIRED");
33:         using var result = await resets.ResetAsync(reference!, sessionId, userId, Convert.FromHexString(tag[1..^1]), ct);
34:         switch (result.Status)
35:         {
36:             case ConditionalCredentialResetStatus.Success:
37:                 Response.Headers.ETag = "\"" + Convert.ToHexString(result.Version!) + "\"";
38:                 return Ok(new { temporaryCredential = new InitialCredentialHandoffDto(result.Credential!.TakePasswordOnce(), result.Credential.ExpiresAtUtc) });
39:             case ConditionalCredentialResetStatus.SessionInvalid: return Failure(401, "SESSION_INVALID");
40:             case ConditionalCredentialResetStatus.ReauthenticationRequired: return Failure(403, "REAUTHENTICATION_REQUIRED");
41:             case ConditionalCredentialResetStatus.SelfDenied:
42:             case ConditionalCredentialResetStatus.Forbidden: return Failure(403, "RESET_NOT_ALLOWED");
43:             case ConditionalCredentialResetStatus.NotFound: return Failure(404, "USER_NOT_FOUND");
44:             case ConditionalCredentialResetStatus.Inactive: return Failure(409, "TARGET_NOT_ACTIVE");
45:             case ConditionalCredentialResetStatus.Stale: return Failure(412, "STALE_USER_VERSION");
46:             case ConditionalCredentialResetStatus.LastAdministrator: return Failure(409, "LAST_DURABLE_ADMIN");
47:             case ConditionalCredentialResetStatus.SerializationUnavailable: return Failure(503, "GLOBAL_ADMIN_SERIALIZATION_UNAVAILABLE");
48:             default: throw new InvalidOperationException("Unknown credential reset result.");
49:         }
50:     }
51: 
52:     private ObjectResult Failure(int status, string code) => Problem(statusCode: status, title: code,
53:         extensions: new Dictionary<string, object?> { ["code"] = code });
