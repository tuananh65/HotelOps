15:     public async Task LogoutAsync(SessionReference? reference, CancellationToken cancellationToken = default)
16:     {
17:         cancellationToken.ThrowIfCancellationRequested();
18:         if (reference is null) return; // No usable credential; middleware already established this state.
19:         await RunAsync(async () =>
20:         {
21:             await using var transaction = await db.Database.BeginTransactionAsync(IsolationLevel.ReadCommitted, cancellationToken);
22:             // False means SQL confirmed missing/already revoked; both are idempotent success.
23:             await sessions.RevokeSessionAsync(reference, cancellationToken);
24:             await transaction.CommitAsync(cancellationToken);
25:             return true;
26:         }, cancellationToken);
27:     }

29:     public async Task<ChangePasswordResult> ChangePasswordAsync(SessionReference reference, long sessionId,
30:         ChangePasswordRequest request, CancellationToken cancellationToken = default)
31:     {
32:         ArgumentNullException.ThrowIfNull(reference);
33:         ArgumentNullException.ThrowIfNull(request);
34:         cancellationToken.ThrowIfCancellationRequested();
35:         if (!PasswordChangePolicy.HasValidInput(request)) return new(ChangePasswordStatus.ValidationFailed);
36:         if (!PasswordChangePolicy.IsStrongEnough(request.NewPassword)) return new(ChangePasswordStatus.PolicyViolation);
37: 
38:         try
39:         {
40:             return await RunAsync(async () =>
41:             {
42:                 await using var transaction = await db.Database.BeginTransactionAsync(IsolationLevel.ReadCommitted, cancellationToken);
43:                 // Validate acquires users UPDLOCK,HOLDLOCK before the session lock; keep both until commit.
44:                 var valid = await sessions.ValidateSessionAsync(reference, cancellationToken);
45:                 if (!Matches(valid, reference, sessionId)) return new ChangePasswordResult(ChangePasswordStatus.SessionInvalid);
46:                 var user = await db.Users.AsNoTracking().SingleAsync(u => u.UserId == reference.UserId, cancellationToken);
47:                 if (passwords.VerifyPassword(user, user.PasswordHash, request.CurrentPassword) == PasswordVerificationResult.Failed)
48:                     return new ChangePasswordResult(ChangePasswordStatus.CurrentPasswordInvalid);
49:                 if (passwords.VerifyPassword(user, user.PasswordHash, request.NewPassword) != PasswordVerificationResult.Failed)
50:                     return new ChangePasswordResult(ChangePasswordStatus.PolicyViolation);
51: 
52:                 var newHash = passwords.HashPassword(user, request.NewPassword);
53:                 // Time can advance while hashing: expired temporary/session cannot be promoted.
54:                 var latest = await sessions.ValidateSessionAsync(reference, cancellationToken);
55:                 if (!Matches(latest, reference, sessionId)) return new ChangePasswordResult(ChangePasswordStatus.SessionInvalid);
56:                 var current = await db.Users.AsNoTracking().SingleAsync(u => u.UserId == reference.UserId, cancellationToken);
57:                 if (!string.Equals(current.PasswordHash, user.PasswordHash, StringComparison.Ordinal)
58:                     || current.TemporaryPasswordExpiresAt != user.TemporaryPasswordExpiresAt
59:                     || current.AccountStatus != user.AccountStatus || current.DeletedAt != user.DeletedAt)
60:                     return new ChangePasswordResult(ChangePasswordStatus.StateChanged);
61: 
62:                 var now = clock.GetUtcNow().UtcDateTime;
63:                 if (latest.Session!.ExpiresAtUtc <= now || latest.Session.AbsoluteExpiresAtUtc <= now
64:                     || current.TemporaryPasswordExpiresAt <= now)
65:                     return new ChangePasswordResult(ChangePasswordStatus.SessionInvalid);
66:                 var changed = await db.Users.Where(u => u.UserId == user.UserId && u.PasswordHash == user.PasswordHash
67:                         && u.TemporaryPasswordExpiresAt == user.TemporaryPasswordExpiresAt
68:                         && u.AccountStatus == user.AccountStatus && u.DeletedAt == null)
69:                     .ExecuteUpdateAsync(s => s.SetProperty(u => u.PasswordHash, newHash)
70:                         .SetProperty(u => u.TemporaryPasswordExpiresAt, (DateTime?)null)
71:                         .SetProperty(u => u.FailedLoginCount, 0).SetProperty(u => u.LockedUntil, (DateTime?)null)
77:                 // Internal proof can only be minted after verified current password + valid session,
78:                 // credential mutation and revoke-all, while retaining the same user lock/transaction.
79:                 var proof = new CredentialLockoutResult(CredentialLockoutStatus.Verified)
80:                 { Proof = new CredentialVerificationProof(user.UserId, transaction.TransactionId, newHash, null) };
81:                 var created = await sessions.CreateSessionAsync(proof, cancellationToken);
82:                 if (created.Session.IsRestricted || created.Session.SessionId == sessionId
88:                     new { authenticationEvent = "PASSWORD_CHANGE", revoked_session_count = revoked, temporary_password_expires_at = (DateTime?)null },
89:                     "password_hash", "temporary_password_expires_at", "failed_login_count", "locked_until", "updated_at");
90:                 audit.ActorUserId = user.UserId;
91:                 db.AuditLogs.Add(audit);
92:                 try { await db.SaveChangesAsync(cancellationToken); }
93:                 finally { db.Entry(audit).State = EntityState.Detached; }
94:                 await transaction.CommitAsync(cancellationToken);
95:                 return new ChangePasswordResult(ChangePasswordStatus.Succeeded,
96:                     new(user.UserId.ToString(CultureInfo.InvariantCulture), user.Username, user.FullName), created);
97:             }, cancellationToken);
98:         }
99:         catch (DbUpdateConcurrencyException) { return new(ChangePasswordStatus.StateChanged); }
100:     }

102:     private static bool Matches(SessionValidationResult result, SessionReference reference, long sessionId)
103:         => result.IsValid && result.Session is { } s && s.SessionId == sessionId && s.UserId == reference.UserId
104:             && s.IsRestricted == reference.IsRestricted && s.CreatedAtUtc == reference.CreatedAtUtc;
