25:         if (db.ChangeTracker.HasChanges() || db.ChangeTracker.Entries<Entities.UserSession>().Any()
26:             || db.ChangeTracker.Entries<Entities.User>().Any())
28:

30:             $"SELECT * FROM dbo.users WITH (UPDLOCK, HOLDLOCK) WHERE user_id = {userId}")
31:             .AsNoTracking().ToListAsync(cancellationToken);
32:         if (users.Count != 1)
33:             throw new InvalidOperationException("Credential target was not found.");
