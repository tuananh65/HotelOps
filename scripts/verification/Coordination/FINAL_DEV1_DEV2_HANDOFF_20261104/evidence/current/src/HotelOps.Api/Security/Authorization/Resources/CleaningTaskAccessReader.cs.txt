8:     IReadOnlyList<CleaningTaskAssignee> Assignments);

11: {

15: public sealed class EfCleaningTaskAccessReader(HotelOpsDbContext db) : ICleaningTaskAccessReader
16: {
17:     public async Task<CleaningTaskAccess?> ReadAsync(long cleaningTaskId, CancellationToken cancellationToken)
18:     {
19:         var task = await db.CleaningTasks.AsNoTracking().Where(t => t.CleaningTaskId == cleaningTaskId)
20:             .Select(t => new
21:             {
22:                 t.Room.BranchId,
23:                 RoomActive = t.Room.IsActive,
24:                 t.Status,
25:                 Assignments = t.CleaningAssignments.Select(a => new CleaningTaskAssignee(
26:                     a.AssigneeUserId, a.AssignmentStatus, a.AssignedAt, a.EndedAt)).ToList()
27:             }).SingleOrDefaultAsync(cancellationToken);
28:         return task is null ? null : new CleaningTaskAccess(task.BranchId, task.RoomActive, task.Status, task.Assignments);
29:     }
30: }
