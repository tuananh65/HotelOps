4: namespace HotelOps.Api.Security.Authorization;
5: 
6: public static class PermissionAuthorizationRegistration
7: {
8:     public static IServiceCollection AddHotelOpsPermissionAuthorization(this IServiceCollection services)
9:     {
10:         services.AddAuthorization(options =>
11:         {
13:             // Explicit permission policies retain their own requirements and scope checks.
14:             options.FallbackPolicy = new AuthorizationPolicyBuilder(HotelOpsCookieTicket.Scheme)
15:                 .RequireAuthenticatedUser().Build();
16:         });
17:         services.AddHttpContextAccessor();
18:         services.TryAddScoped<IPermissionPolicyCatalog, EfPermissionPolicyCatalog>();
19:         services.TryAddEnumerable(ServiceDescriptor.Scoped<IAuthorizationHandler, PermissionAuthorizationHandler>());
20:         services.Replace(ServiceDescriptor.Singleton<IAuthorizationPolicyProvider, PermissionPolicyProvider>());
21:         return services;
22:     }
23: }
