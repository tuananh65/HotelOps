39:             throw new ValidationException("Status must be OPEN, IN_PROGRESS, RESOLVED, CLOSED, or CANCELLED.");

46:             var allowed = (incident.Status, status) switch
47:             {
48:                 ("OPEN", "IN_PROGRESS" or "CANCELLED") => true,
49:                 ("IN_PROGRESS", "RESOLVED" or "CANCELLED") => true,
52:             };

54:                 throw new ValidationException($"Transition from {incident.Status} to {status} is not allowed.");

60:                 throw new ValidationException("A reason is required when returning a resolved incident for rework.");
61:

65:             var previousClosedAt = incident.ClosedAt;
