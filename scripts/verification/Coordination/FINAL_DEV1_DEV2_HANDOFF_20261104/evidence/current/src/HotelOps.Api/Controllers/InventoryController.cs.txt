16:     private async Task<IActionResult> Execute<T>(Func<long, Task<T>> operation, Func<T, string>? location = null, object? input = null)
17:     {
18:         try
19:         {
20:             if (!current.TryGetCurrentUser(out var caller)) return Unauthorized();
21:             if (caller.IsRestricted) return Forbid(HotelOpsCookieTicket.Scheme);
22:             if (!await access.Authorize(ControllerContext.ActionDescriptor.ActionName, RouteData.Values, input, HttpContext.RequestAborted)) return NotFound();
23:             if (HttpMethods.IsGet(Request.Method)) service.RestrictReads(await access.ReadWarehouses(HttpContext.RequestAborted));
24:             var result = await operation(caller.UserId);
25:             if (result is null) return NotFound();
26:             return location is null ? Ok(result) : Created(location(result), result);
27:         }
28:         catch (InventoryAccessException e) { return e.Status == 401 ? Unauthorized() : Forbid(HotelOpsCookieTicket.Scheme); }
29:         catch (ValidationException exception)
30:         {
31:             return Problem(statusCode: StatusCodes.Status400BadRequest, title: "Inventory validation failed", detail: exception.Message);
32:         }
33:     }
