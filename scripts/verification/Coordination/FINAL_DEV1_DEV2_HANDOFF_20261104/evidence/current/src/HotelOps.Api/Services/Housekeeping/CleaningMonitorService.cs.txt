110:     private async Task<CleaningMonitorResult> ReadAsync(SessionReference ticket, long sessionId, long branchId,
111:         Func<CancellationToken, Task<CleaningMonitorResult>> read, CancellationToken ct)
112:     {
113:         if (!current.TryGetCurrentUser(out var actor) || actor.IsRestricted || actor.UserId != ticket.UserId
114:             || actor.SessionId != sessionId || ticket.IsRestricted) return new(CleaningMonitorStatus.SessionInvalid);
115:         if (db.Database.CurrentTransaction is not null) throw new InvalidOperationException("Monitor owns its read transaction.");
116:         try
117:         {
118:             // Retain session/permission read locks through count, page and detail; no business mutations.
119:             await using var tx = await db.Database.BeginTransactionAsync(IsolationLevel.Serializable, ct);
120:             if (await Gate() is { } denied) return new(denied);
121:             var result = await read(ct);
122:             if (await Gate() is { } changed) return new(changed);
123:             await tx.CommitAsync(ct);
124:             return result;
125:         }
126:         catch (SqlException error) when (!ct.IsCancellationRequested && error.Number is -2 or 1205 or 1222 or 207 or 208)
127:         { return new(CleaningMonitorStatus.Unavailable); }

129:         async Task<CleaningMonitorStatus?> Gate()
130:         {
131:             var valid = await sessions.ValidateSessionAsync(ticket, ct);
132:             if (!valid.IsValid || valid.Session?.SessionId != sessionId || valid.Session.IsRestricted)
133:                 return CleaningMonitorStatus.SessionInvalid;
134:             // Supervisor monitor requires an actual branch/global view grant. OWN is not promoted to branch access.
135:             var access = await permissions.EvaluateBranchAsync("CLEANING_VIEW", branchId, ct);
