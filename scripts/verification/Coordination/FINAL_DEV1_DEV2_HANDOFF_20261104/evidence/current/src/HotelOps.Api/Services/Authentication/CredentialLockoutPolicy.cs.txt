3: /// <summary>Pure state transitions; storage/serialization belongs to the service.</summary>
4: public static class CredentialLockoutPolicy
5: {
6:     public const int FailureThreshold = 5;
7:     public static readonly TimeSpan LockDuration = TimeSpan.FromMinutes(30);
8: 
9:     public static CredentialLockoutTransition Evaluate(
10:         int failedCount, DateTime? lockedUntil, bool passwordVerified,
11:         bool eligibleForLogin, DateTime utcNow)
12:     {
20:             // Wrong passwords cannot discover the lock or extend its deadline.
21:             var status = passwordVerified && eligibleForLogin
22:                 ? CredentialLockoutStatus.Locked : CredentialLockoutStatus.InvalidCredentials;
23:             return new(status, failedCount, lockedUntil, false);
24:         }
25: 
27:         {
28:             return eligibleForLogin
29:                 ? new(CredentialLockoutStatus.Verified, 0, null, false)
30:                 : new(CredentialLockoutStatus.InvalidCredentials, failedCount, lockedUntil, false);
31:         }
32: 
33:         // An expired lock starts a fresh failure cycle (including now == deadline).
34:         var count = lockedUntil.HasValue ? 1 : Math.Min(failedCount, FailureThreshold - 1) + 1;
35:         var startsLock = count >= FailureThreshold;
36:         return new(CredentialLockoutStatus.InvalidCredentials, count,
37:             startsLock ? utcNow.Add(LockDuration) : null, startsLock);
38:     }
39: }
40: 
41: public sealed record CredentialLockoutTransition(
42:     CredentialLockoutStatus Status, int FailedLoginCount, DateTime? LockedUntil,
43:     bool LockStarted);
