57:     // Scheduled foundation primitive. No public endpoint or implicit attendance->Workday mapping.
58:     public Task<Workday> StartScheduledWorkdayAsync(long actorUserId, long branchId, long shiftAssignmentId,
59:         CancellationToken ct) => RunAsync(async () =>
60:     {
61:         RequireLocks(actorUserId);
62:         if (branchId <= 0 || shiftAssignmentId <= 0) throw Failure("TIME_STATE_CONFLICT");
63:         var now = await NowAsync(ct);
64:         await EligibleAsync(actorUserId, branchId, now, ct);
66:         if (await Open(actorUserId).AnyAsync(ct)) throw Failure("TIME_STATE_CONFLICT");
67:         var assignment = await db.ShiftAssignments.FromSqlInterpolated(
68:             $"SELECT * FROM dbo.shift_assignments WITH(UPDLOCK,HOLDLOCK) WHERE shift_assignment_id={shiftAssignmentId}")
69:             .AsNoTracking().SingleOrDefaultAsync(ct);
70:         // ASSIGNED is the actual legacy schema default, not an invented SRS status.
71:         if (assignment is null || assignment.UserId != actorUserId || assignment.AssignmentStatus != "ASSIGNED"
72:             || assignment.AssignedAt > now) throw Failure("TIME_SCOPE_DENIED");
73:         var shift = await db.WorkShifts.FromSqlInterpolated(
74:             $"SELECT * FROM dbo.work_shifts WITH(UPDLOCK,HOLDLOCK) WHERE shift_id={assignment.ShiftId}")
75:             .AsNoTracking().SingleOrDefaultAsync(ct);
76:         var today = DateOnly.FromDateTime(TimeZoneInfo.ConvertTimeFromUtc(now, HotelTimezone));
77:         if (shift is null || shift.BranchId != branchId || shift.Status != "PLANNED" || shift.ShiftDate != today
78:             || shift.ScheduledEndAt < shift.ScheduledStartAt) throw Failure("TIME_STATE_CONFLICT");
79:         if (!await db.ShiftTemplates.FromSqlInterpolated(
80:                 $"SELECT * FROM dbo.shift_templates WITH(HOLDLOCK) WHERE shift_template_id={shift.ShiftTemplateId}")
81:                 .AsNoTracking().AnyAsync(x => x.ShiftTemplateId == shift.ShiftTemplateId
82:                 && x.BranchId == branchId && x.IsActive, ct)
83:             || !await db.Departments.FromSqlInterpolated(
84:                 $"SELECT * FROM dbo.departments WITH(HOLDLOCK) WHERE department_id={shift.DepartmentId}")
85:                 .AsNoTracking().AnyAsync(x => x.DepartmentId == shift.DepartmentId
86:                 && x.BranchId == branchId && x.IsActive, ct)) throw Failure("TIME_SCOPE_DENIED");
87:         now = await NowAsync(ct);
88:         var day = new Workday { UserId = actorUserId, BranchId = branchId, DepartmentId = shift.DepartmentId,
89:             WorkDate = shift.ShiftDate, ShiftAssignmentId = shiftAssignmentId, IsUnscheduled = false,
90:             Status = "WORKING", StartedAt = now };
91:         db.Workdays.Add(day);
92:         await db.SaveChangesAsync(ct);
93:         Audit(branchId, actorUserId, "workdays", day.WorkdayId, "CREATE", null,
94:             new { day.UserId, day.Status, day.ShiftAssignmentId, day.StartedAt }, now);
95:         await db.SaveChangesAsync(ct);
96:         return day;
97:     });

203:     private async Task EligibleAsync(long userId, long branchId, DateTime now, CancellationToken ct)
204:     {
205:         var user = await db.Users.AsNoTracking().Where(x => x.UserId == userId)

208:             throw Failure("TIME_ACTOR_INVALID");

211:             || !await db.UserBranchAccesses.FromSqlInterpolated(
212:                 $"SELECT * FROM dbo.user_branch_access WITH(HOLDLOCK) WHERE user_id={userId} AND branch_id={branchId}")
213:                 .AsNoTracking().AnyAsync(x => x.UserId == userId && x.BranchId == branchId
214:                 && x.GrantedAt <= now && x.RevokedAt == null, ct)) throw Failure("TIME_SCOPE_DENIED");

216:     private async Task SourceAsync(WorkTimeOperation op, DateTime now, CancellationToken ct)
217:     {
218:         // Additional authoritative consistency checks. Caller still owns source locks/permission/transition.
219:         bool valid;
220:         if (op.WorkItemKind == "CLEANING")
221:             valid = await db.CleaningTasks.AsNoTracking().AnyAsync(x => x.CleaningTaskId == op.WorkItemId
222:                 && x.Status == "IN_PROGRESS" && x.Room.BranchId == op.BranchId && x.Room.IsActive
223:                 && x.CleaningAssignments.Count(a => a.AssignmentStatus == "ACTIVE" && a.EndedAt == null && a.AssignedAt <= now) == 1
224:                 && x.CleaningAssignments.Any(a => a.AssignmentStatus == "ACTIVE" && a.EndedAt == null
225:                     && a.AssignedAt <= now && a.AssigneeUserId == op.OwnerUserId), ct);
226:         else if (op.WorkItemKind == "REWORK")
227:             valid = await db.CleaningReworks.AsNoTracking().AnyAsync(x => x.ReworkId == op.WorkItemId
228:                 && x.Status == "IN_PROGRESS" && x.CompletedAt == null && x.AssigneeUserId == op.OwnerUserId
229:                 && x.CleaningTask.Room.BranchId == op.BranchId && x.CleaningTask.Room.IsActive
230:                 && x.Inspection.RoomId == x.CleaningTask.RoomId && x.Inspection.CleaningTaskId == x.CleaningTaskId, ct);
231:         else
232:             valid = await db.OperationalTasks.AsNoTracking().AnyAsync(x => x.TaskId == op.WorkItemId && x.BranchId == op.BranchId
233:                 && x.Status == "IN_PROGRESS" && x.Department.BranchId == op.BranchId && x.Department.IsActive
234:                 && x.OperationalTaskAssignments.Count(a => a.AssignmentStatus == "ACTIVE" && a.EndedAt == null && a.AssignedAt <= now) == 1
235:                 && x.OperationalTaskAssignments.Any(a => a.AssignmentStatus == "ACTIVE" && a.EndedAt == null
236:                     && a.AssignedAt <= now && a.AssigneeId == op.OwnerUserId), ct);
237:         if (!valid) throw Failure("TIME_SCOPE_DENIED");
238:     }
